Solution-
cat > hello.c <<'EOF'

int main() {
    printf("Hello from C\n");
    return 0;
}
EOF
cat > build.sh <<'EOF'
#!/bin/bash
gcc hello.c -o hello
EOF
chmod 777 build.sh
./build.sh
./hello
